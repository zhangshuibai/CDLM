# Training dependencies for the Open-dCoder-0.5B CDLM / MDLM runs
# (Python 3.11.13, Linux x86_64 with glibc >= 2.28, torch built for CUDA 12.4, A100 GPUs).
# constraints.txt pins every other package, transitive dependencies included, to a verified
# environment; install with -c constraints.txt (below).
#
# Versions. The pins below are the environment of the multi-seed runs (seeds 1234 and 2025)
# and the clean-token-weight run, which used exactly the veomni/ and tasks/ code in this
# directory. The headline seed-42 runs that produced the released CDLM-0.5B and MDLM-0.5B
# checkpoints used the same version of every package listed here except
#   transformers==4.54.1  tokenizers==0.21.4  datasets==3.6.0
# and a slightly older copy of the code (it differs in logging and timeouts only; see NOTICE).
# With the pins below, the headline MDLM run's data stream was reproduced for 2001 steps, with
# losses bitwise equal at steps 1, 2 and 2001. To use the headline runs' versions instead,
# install those three pins after this file (not re-run with this code snapshot).
#
# Install order:
#   pip install torch==2.5.0 --index-url https://download.pytorch.org/whl/cu124
#   pip install https://github.com/Dao-AILab/flash-attention/releases/download/v2.7.4.post1/flash_attn-2.7.4.post1+cu12torch2.5cxx11abiFALSE-cp311-cp311-linux_x86_64.whl
#   pip install -r requirements.txt -c constraints.txt
#   pip check
#
# flash-attn: the URL above is the official prebuilt wheel for torch 2.5, CUDA 12,
# Python 3.11 and the pre-C++11 ABI that the torch 2.5.0 wheels use; it needs no
# CUDA toolkit. `pip install flash-attn==2.7.4.post1 --no-build-isolation` does not
# work everywhere: its setup.py checks nvcc before it looks for a prebuilt wheel
# and aborts when the nvcc on PATH is older than CUDA 11.7 (for example a system
# CUDA 10.1). Building from source needs nvcc >= 11.7, ninja, packaging and a lot
# of RAM (MAX_JOBS=4). Either way it must be installed after torch.

torch==2.5.0
triton==3.1.0
# triton 3.1.0 imports setuptools at import time (triton/runtime/build.py) but does not
# declare it; torch 2.5.0 declares it only for Python >= 3.12.
setuptools==78.1.1
flash-attn==2.7.4.post1
# Required: the MDM / mixture losses in modeling_qwen2.py use
# LigerFusedLinearCrossEntropyLoss and raise if liger-kernel is missing.
# In 0.5.8 its backward with reduction="none" multiplies every position's gradient by the
# upstream gradient of the first position (row 0) only, so the per-token loss weights do not
# reach the gradient (README.md, "Effective gradient"). The released 0.5B checkpoints were trained this way; another
# liger-kernel version can change the gradients.
liger-kernel==0.5.8

# 4.57.0 was later yanked on PyPI ("Error in the setup causing installation issues");
# the exact pin still installs, with a warning.
transformers==4.57.0
tokenizers==0.22.2
safetensors==0.6.2
huggingface-hub==0.34.4
accelerate==1.11.0

# datasets 3.6.0 (headline runs) and 4.2.0 give the same streaming shuffle / shard split
# for this data (see the 2001-step check above); keep the pin so that stays true.
datasets==4.2.0
fsspec==2024.6.1
pyarrow==22.0.0
torchdata==0.11.0

numpy==2.3.2
packaging==25.0
# Imported whenever veomni.models is imported, i.e. by tasks/train_torch.py and
# tools/convert_dcp_to_hf.py (veomni/models/seed_omni/decoder/movqgan/processing_movqgan.py,
# and transformers.image_utils.PILImageResampling in veomni/models/transformers/movqgan/).
pillow==11.3.0

wandb==0.23.0

# Only needed with --train.ckpt_manager=bytecheckpoint (the yaml default);
